<?php
$home = getenv('HOME');
require $home . '/www/akb-66.ru/api/lib.php';
echo "token_len=", strlen((string)cfg('max_token')), "\n";
$me = max_api('GET', '/me');
echo "me http=", $me['_http'] ?? '-', " bot=", $me['name'] ?? ($me['first_name'] ?? '-'), " @", $me['username'] ?? '-', "\n";
$u = max_api('GET', '/updates', ['limit' => 100, 'timeout' => 0]);
echo "updates http=", $u['_http'] ?? '-', " n=", count($u['updates'] ?? []), "\n";
$chats = [];
foreach (($u['updates'] ?? []) as $x) {
    $m = $x['message'] ?? null;
    $cid = $m['recipient']['chat_id'] ?? ($x['chat_id'] ?? null);
    $who = $m['sender']['name'] ?? ($x['user']['name'] ?? '?');
    if ($cid) $chats[$cid] = ($x['update_type'] ?? '?') . ' от ' . $who;
}
foreach ($chats as $c => $d) echo "chat ", $c, " : ", $d, "\n";
if (count($chats) === 1) {
    $cid = array_key_first($chats);
    $f = DATA . '/config.php'; $s = file_get_contents($f);
    $s2 = preg_replace("/'max_chat_id'\s*=>\s*'[^']*'/", "'max_chat_id'  => '" . $cid . "'", $s, 1);
    if ($s2 && $s2 !== $s) { file_put_contents($f, $s2, LOCK_EX); echo "chat_id saved\n"; }
    $r = max_api('POST', '/messages', ['chat_id' => $cid], ['text' => "66 Вольт: уведомления о заказах с сайта akb-66.ru подключены. Сюда будут приходить новые заказы и заявки."]);
    echo "test msg http=", $r['_http'] ?? '-', "\n";
}
