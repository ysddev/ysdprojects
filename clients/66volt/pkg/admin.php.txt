// Все параметры аккумулятора из формы (общие для «Добавить» и «Изменить»)
function item_from_post(array $x): array {
    $x['b']  = clean($_POST['b'] ?? $x['b'] ?? '', 40) ?: ($x['b'] ?? '');
    $x['t']  = clean($_POST['t'] ?? $x['t'] ?? '', 120) ?: ($x['t'] ?? '');
    $x['c']  = (int)($_POST['c'] ?? $x['c'] ?? 0);
    $x['a']  = ((int)($_POST['a2'] ?? 0)) ?: null;
    $L = (int)($_POST['L'] ?? 0); $W = (int)($_POST['W'] ?? 0); $H = (int)($_POST['H'] ?? 0);
    $x['d']  = ($L && $W && $H) ? [$L, $W, $H] : null;
    $x['p']  = in_array($_POST['pol'] ?? '', ['R', 'L', '3', '4'], true) ? $_POST['pol'] : ($x['p'] ?? '?');
    $x['k']  = in_array($_POST['k'] ?? '', ['E', 'A', 'T'], true) ? $_POST['k'] : ($x['k'] ?? 'E');
    $x['tm'] = in_array($_POST['tm'] ?? '', TM_LIST, true) ? $_POST['tm'] : ($x['tm'] ?? 'Стандартные');
    $x['te'] = in_array($_POST['te'] ?? '', TE_LIST, true) ? $_POST['te'] : ($x['te'] ?? 'Стандарт');
    $x['lo'] = !empty($_POST['lo']) ? 1 : 0;
    $st = trim((string)($_POST['s'] ?? ''));
    $x['s']  = $st === '' ? null : max(0, (int)$st);
    $x['pr'] = max(0, (int)preg_replace('/\D/', '', (string)($_POST['pr'] ?? ($x['pr'] ?? 0))));
    $x['mf'] = clean($_POST['mf'] ?? ($x['mf'] ?? ''), 80);
    $x['h']  = !empty($_POST['h']) ? 1 : 0;
    return $x;
}

function handle_post(string $p): void {
    $a = $_POST['a'] ?? '';
    if ($a === 'status') {
        $allowed = ['new', 'awaiting_payment', 'paid', 'done', 'cancelled', 'request', 'payment_error', 'rejected', 'refunded'];
        $s = $_POST['status'] ?? '';
        if (in_array($s, $allowed, true)) db()->prepare('UPDATE orders SET status=? WHERE id=?')->execute([$s, (int)$_POST['id']]);
        go('p=orders');
    }
    if ($a === 'save' || $a === 'photo' || $a === 'nophoto' || $a === 'add') {
        $items = catalog();
        if ($a === 'add') {
            $ids = array_map(fn($x) => (int)substr($x['id'], 4), $items);
            $new = item_from_post(['id' => sprintf('AKB-%04d', max($ids ?: [0]) + 1)]);
            if ($new['b'] && $new['c'] && $new['pr']) { $items[] = $new; save_catalog($items); }
            go('p=products&q=' . urlencode($new['id']));
        }
        $id = (string)($_POST['id'] ?? '');
        foreach ($items as &$x) {
            if ($x['id'] !== $id) continue;
            if ($a === 'save') {
                $x = item_from_post($x);
            } elseif ($a === 'photo' && !empty($_FILES['img']['tmp_name']) && $_FILES['img']['size'] < 6_000_000) {
                $info = @getimagesize($_FILES['img']['tmp_name']);
                $ext = ['image/jpeg' => 'jpg', 'image/png' => 'png', 'image/webp' => 'webp'][$info['mime'] ?? ''] ?? null;
                if ($ext) {
                    @mkdir(ROOT . '/uploads', 0755);
                    $fn = 'uploads/' . strtolower($x['id']) . '-' . bin2hex(random_bytes(3)) . '.' . $ext;
                    if (move_uploaded_file($_FILES['img']['tmp_name'], ROOT . '/' . $fn)) {
                        if (!empty($x['img']) && str_starts_with($x['img'], 'uploads/')) @unlink(ROOT . '/' . $x['img']);
                        $x['img'] = $fn;
                    }
                }
            } elseif ($a === 'nophoto') {
                if (!empty($x['img']) && str_starts_with($x['img'], 'uploads/')) @unlink(ROOT . '/' . $x['img']);
                unset($x['img']);
            }
        }
        unset($x);
        save_catalog($items);
        go((string)($_POST['back'] ?? 'p=products') . '#' . urlencode($id));
    }
    if ($a === 'testmail') {
        $ok = send_mail('Проверка почты — 66 Вольт', 'Если вы видите это письмо, уведомления о заказах будут приходить сюда.');
        $_SESSION['flash'] = $ok ? 'Письмо отправлено на ' . cfg('shop_email') . ' — проверьте почту (и папку «Спам»).' : 'Не удалось отправить письмо: функция mail() на хостинге не сработала.';
        go('p=diag');
    }
    if ($a === 'testmax') {
        $_SESSION['flash'] = send_max('Проверка уведомлений — 66 Вольт') ? 'Сообщение в Max отправлено.' : 'Max не настроен или токен неверный.';
        go('p=diag');
    }
    go('p=' . $p);
}

// ---------------- страницы ----------------
function nav(string $cur): string {
    $l = ['orders' => 'Заказы и заявки', 'products' => 'Товары и цены', 'diag' => 'Проверка'];
    $s = '<nav>'; foreach ($l as $k => $t) $s .= '<a href="?p=' . $k . '"' . ($k === $cur ? ' class="on"' : '') . '>' . $t . '</a>';
    return $s . '<a href="../" target="_blank">Сайт ↗</a><a href="?logout=1">Выйти</a></nav>';

function orders(): void {
    $rows = db()->query('SELECT * FROM orders ORDER BY id DESC LIMIT 300')->fetchAll();
    $sts = ['new', 'awaiting_payment', 'paid', 'done', 'cancelled', 'payment_error', 'rejected', 'refunded', 'request'];
    $o = nav('orders') . '<div class="box"><h1>Заказы и заявки</h1>';
    if (!$rows) $o .= '<p class="muted">Пока пусто. Новые заказы и заявки появятся здесь и придут на почту.</p>';
    foreach ($rows as $r) {
        $items = json_decode($r['items'], true) ?: [];
        $o .= '<div class="ord st-' . h($r['status']) . '"><div class="oh"><b>' . ($r['kind'] === 'request' ? 'Заявка на подбор' : 'Заказ') . ' №' . (int)$r['id'] . '</b><span>' . h($r['created']) . '</span>';
        $o .= '<form method="post"><input type="hidden" name="csrf" value="' . csrf() . '"><input type="hidden" name="a" value="status"><input type="hidden" name="id" value="' . (int)$r['id'] . '"><select name="status" onchange="this.form.submit()">';
        foreach ($sts as $s) $o .= '<option value="' . $s . '"' . ($s === $r['status'] ? ' selected' : '') . '>' . h(status_name($s)) . '</option>';
        $o .= '</select></form></div>';
        if ($r['kind'] === 'request') {
            $o .= '<p>Авто: <b>' . h($r['note']) . '</b><br>Телефон: <a href="tel:' . h($r['phone']) . '">' . h($r['phone']) . '</a></p>';
        } else {
            $o .= '<ul>'; foreach ($items as $it) $o .= '<li>' . h($it['name']) . ' × ' . (int)$it['q'] . ' — ' . rub((int)$it['sum']) . '</li>'; $o .= '</ul>';
            $o .= '<p><b>Итого: ' . rub((int)$r['total']) . '</b><br>' . h($r['name']) . ', <a href="tel:' . h($r['phone']) . '">' . h($r['phone']) . '</a>' . ($r['email'] ? ', ' . h($r['email']) : '') .
                '<br>' . ($r['dlv'] === 'courier' ? 'Курьер: ' . h($r['addr']) : 'Самовывоз из магазина') . ' · ' . ($r['pay'] === 'online' ? 'оплата на сайте' : 'оплата при получении') .
                ($r['note'] ? '<br><span class="muted">' . h($r['note']) . '</span>' : '') . '</p>';
        }
        $o .= '</div>';
    }
    page('Заказы', $o . '</div>');
}

function products(): void {
    $q = trim((string)($_GET['q'] ?? '')); $ql = mb_strtolower($q);
    $br = trim((string)($_GET['b'] ?? ''));
    $hid = !empty($_GET['hid']);
    $all = catalog();
    $brands = [];
    foreach ($all as $x) $brands[$x['b']] = ($brands[$x['b']] ?? 0) + 1;
    arsort($brands);
    $list = array_values(array_filter($all, function ($x) use ($ql, $br, $hid) {
        if ($br !== '' && $x['b'] !== $br) return false;
        if ($hid && empty($x['h'])) return false;
        return $ql === '' || str_contains(mb_strtolower($x['id'] . ' ' . $x['b'] . ' ' . $x['t'] . ' ' . ($x['mf'] ?? '') . ' ' . $x['c'] . 'ач'), $ql);
    }));
    $per = 40; $pages = max(1, (int)ceil(count($list) / $per));
    $pg = min($pages, max(1, (int)($_GET['pg'] ?? 1)));
    $show = array_slice($list, ($pg - 1) * $per, $per);
    $base = ['p' => 'products'] + ($q !== '' ? ['q' => $q] : []) + ($br !== '' ? ['b' => $br] : []) + ($hid ? ['hid' => 1] : []);
    $back = http_build_query($base + ($pg > 1 ? ['pg' => $pg] : []));
    $c = csrf();

    $o = nav('products') . '<div class="box"><h1>Товары и цены</h1>';
    $o .= '<div class="chips"><a href="?' . h(http_build_query(['p' => 'products'])) . '"' . ($br === '' && !$hid ? ' class="on"' : '') . '>Все <b>' . count($all) . '</b></a>';
    foreach ($brands as $name => $n) $o .= '<a href="?' . h(http_build_query(['p' => 'products', 'b' => $name])) . '"' . ($br === $name ? ' class="on"' : '') . '>' . h($name) . ' <b>' . $n . '</b></a>';
    $o .= '<a href="?p=products&amp;hid=1"' . ($hid ? ' class="on"' : '') . '>Скрытые</a></div>';
    $o .= '<form class="search"><input type="hidden" name="p" value="products">' . ($br !== '' ? '<input type="hidden" name="b" value="' . h($br) . '">' : '') .
        '<input name="q" value="' . h($q) . '" placeholder="Поиск' . ($br !== '' ? ' в «' . h($br) . '»' : '') . ': модель, ёмкость, артикул (например «60» или «AKB-0123»)"><button>Найти</button></form>';
    $o .= '<p class="muted">' . ($br !== '' ? 'Бренд «' . h($br) . '»: ' : '') . 'найдено ' . count($list) . ($pages > 1 ? ', страница ' . $pg . ' из ' . $pages : '') . '. Нажмите «Изменить», чтобы править любые параметры. Цена — розничная, в рублях.</p>';
    foreach ($show as $x) {
        $o .= '<div class="prod' . (!empty($x['h']) ? ' off' : '') . '" id="' . h($x['id']) . '"><div class="ph">' . (!empty($x['img']) ? '<img src="../' . h($x['img']) . '" alt="">' : '<span>нет фото</span>') . '</div><div class="pf">';
        $o .= '<div class="t1"><b>' . h($x['t']) . '</b> · ' . (int)$x['c'] . ' Ач' . ($x['a'] ? ' · ' . (int)$x['a'] . ' А' : '') . ' · <b>' . number_format((int)$x['pr'], 0, ',', ' ') . ' ₽</b>' . (!empty($x['h']) ? ' · <span class="red">скрыт</span>' : '') . '</div>';
        $o .= '<div class="muted">' . h($x['id']) . ' · ' . h($x['b']) . ($x['d'] ? ' · ' . h(implode('×', $x['d'])) . ' мм' : '') . ' · ' . (['R' => 'обратная', 'L' => 'прямая', '3' => 'тип 3', '4' => 'тип 4'][$x['p']] ?? 'полярность не указана') . '</div>';
        $o .= '<details class="ed"><summary>Изменить</summary><form method="post" class="grid"><input type="hidden" name="csrf" value="' . $c . '"><input type="hidden" name="a" value="save"><input type="hidden" name="id" value="' . h($x['id']) . '"><input type="hidden" name="back" value="' . h($back) . '">' . item_fields($x) . '<button>Сохранить</button></form>';
        $o .= '<div class="row"><form method="post" enctype="multipart/form-data" class="row"><input type="hidden" name="csrf" value="' . $c . '"><input type="hidden" name="a" value="photo"><input type="hidden" name="id" value="' . h($x['id']) . '"><input type="hidden" name="back" value="' . h($back) . '"><input type="file" name="img" accept="image/jpeg,image/png,image/webp" required><button class="ghost">Загрузить фото</button></form>';
        if (!empty($x['img'])) $o .= '<form method="post"><input type="hidden" name="csrf" value="' . $c .
